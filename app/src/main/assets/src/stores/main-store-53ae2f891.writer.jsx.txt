/* Cache Storage as offline-audio-web.test.js fakes it (a put reads the whole body). */
function fakeCaches() {
  const buckets = new Map();
  const bucket = (name) => {
    if (!buckets.has(name)) buckets.set(name, new Map());
    const m = buckets.get(name);
    const k = (r) => (typeof r === 'string' ? r : r.url);
    return {
      keys: async () => [...m.keys()].map((url) => ({ url })),
      match: async (r) => { const e = m.get(k(r)); return e ? new Response(e.bytes.slice(0), { headers: e.headers }) : undefined; },
      put: async (r, res) => { const bytes = new Uint8Array(await res.arrayBuffer()); m.set(k(r), { bytes, headers: new Headers(res.headers) }); },
      delete: async (r) => m.delete(k(r)),
    };
  };
  return { open: async (name) => bucket(name), _buckets: buckets };
}
const define = (obj, name, value) => Object.defineProperty(obj, name, { value, configurable: true, writable: true });

async function blobJson(v) {
  if (v instanceof Blob) return { __blob: true, type: v.type, b64: Buffer.from(await v.arrayBuffer()).toString('base64') };
  if (Array.isArray(v)) return Promise.all(v.map(blobJson));
  if (v && typeof v === 'object') { const o = {}; for (const [k, x] of Object.entries(v)) o[k] = await blobJson(x); return o; }
  return v;
}

/** One database as it stands: each store's shape and every [key, value]. */
function dumpDb(name) {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(name);
    req.onerror = () => reject(req.error);
    req.onsuccess = () => {
      const db = req.result;
      const stores = [...db.objectStoreNames];
      const out = { version: db.version, stores: {} };
      if (!stores.length) { db.close(); resolve(out); return; }
      const tx = db.transaction(stores, 'readonly');
      const raw = {};
      for (const s of stores) {
        const os = tx.objectStore(s);
        raw[s] = { keyPath: os.keyPath, autoIncrement: os.autoIncrement, indexes: [...os.indexNames].map((n) => { const ix = os.index(n); return { name: n, keyPath: ix.keyPath, unique: ix.unique, multiEntry: ix.multiEntry }; }), records: [] };
        os.openCursor().onsuccess = (e) => { const c = e.target.result; if (c) { raw[s].records.push([c.key, c.value]); c.continue(); } };
      }
      tx.oncomplete = async () => {
        db.close();
        for (const [s, d] of Object.entries(raw)) out.stores[s] = { ...d, records: await blobJson(d.records) };
        resolve(out);
      };
      tx.onerror = () => reject(tx.error);
    };
  });
}
